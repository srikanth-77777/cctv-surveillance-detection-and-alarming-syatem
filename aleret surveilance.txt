ALARM

import cv2
import numpy as np
import os
import winsound

print("🚀 Starting Surveillance System with Object Detection and Alarm...")

video_path = r"Z:\Games\video survillance01\video survillance\video_surveillance_project\videos\testing_video.mp4"
prototxt_path = "MobileNetSSD_deploy.prototxt"
model_path = "MobileNetSSD_deploy.caffemodel"

CLASSES = ["background", "aeroplane", "bicycle", "bird", "boat",
           "bottle", "bus", "car", "cat", "chair", "cow", "diningtable",
           "dog", "horse", "motorbike", "person", "pottedplant", "sheep",
           "sofa", "train", "tvmonitor"]

np.random.seed(42)
COLORS = np.random.uniform(0, 255, size=(len(CLASSES), 3))

if not os.path.exists(video_path):
    print(f"❌ Video file not found at {video_path}")
else:
    net = cv2.dnn.readNetFromCaffe(prototxt_path, model_path)
    cap = cv2.VideoCapture(video_path)

    if not cap.isOpened():
        print("❌ Error: Could not open video.")
    else:
        print("📹 Video loaded. Press 'q' to quit.")

        frame_count = 0

        while True:
            ret, frame = cap.read()
            if not ret:
                if cap.get(cv2.CAP_PROP_POS_FRAMES) < cap.get(cv2.CAP_PROP_FRAME_COUNT):
                    print("⚠️ Failed to read frame (unexpected).")
                else:
                    print("✅ Video completed normally.")
                break

            frame_count += 1
            (h, w) = frame.shape[:2]
            blob = cv2.dnn.blobFromImage(cv2.resize(frame, (300, 300)),
                                         0.007843, (300, 300), 127.5)

            net.setInput(blob)
            detections = net.forward()

            detected_something = False

            for i in np.arange(0, detections.shape[2]):
                confidence = detections[0, 0, i, 2]

                if confidence > 0.5:
                    idx = int(detections[0, 0, i, 1])
                    box = detections[0, 0, i, 3:7] * np.array([w, h, w, h])
                    (startX, startY, endX, endY) = box.astype("int")

                    label = f"{CLASSES[idx]}: {confidence * 100:.2f}%"
                    cv2.rectangle(frame, (startX, startY), (endX, endY),
                                  COLORS[idx], 2)
                    y = startY - 15 if startY - 15 > 15 else startY + 15
                    cv2.putText(frame, label, (startX, y),
                                cv2.FONT_HERSHEY_SIMPLEX, 0.5, COLORS[idx], 2)

                    detected_something = True

            if detected_something:
                print(f"🚨 ALERT at frame {frame_count}: Object detected!")
                winsound.Beep(1000, 500)  # Alarm sound
                # Visual Flash - Red Warning Box
                cv2.rectangle(frame, (10, 10), (300, 60), (0, 0, 255), -1)
                cv2.putText(frame, "ALERT: Object Detected!", (20, 45),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)

                # Save frame as image
                alert_folder = "alerts"
                os.makedirs(alert_folder, exist_ok=True)
                alert_filename = os.path.join(alert_folder, f"alert_frame_{frame_count}.jpg")
                cv2.imwrite(alert_filename, frame)
                print(f"📸 Saved alert frame: {alert_filename}")

            cv2.imshow("Video Surveillance - Object Detection + Alarm", frame)

            if cv2.waitKey(25) & 0xFF == ord('q'):
                print("🛑 Stopped by user.")
                break

        cap.release()
        cv2.destroyAllWindows()

print("✅ Surveillance Completed.")
